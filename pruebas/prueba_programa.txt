de_la_abuela enchilada max = 12;

receta salsa cobrar(enchilada tacos, salsa precio) {
    servir tacos * precio + 0.5;
}

receta ayuno principal() {
    tamal orden = "pastor\n";
    pozole letra = 'p';
    picante hambre = con_todo;
    enchilada bits[2];
    pedir(orden);
    cada_taco (pozole c en orden) {
        si_hay (c == letra) { otro_taco; } ni_modo { gritar(c); }
    }
    por_orden (enchilada i = 0; i < 2; i++) {
        bits[i] = (i << 2) >> 1 & ~7 | 3 ^ 1;
    }
    sazonando (hambre && !(max <= 0) || max >= 1) {
        hambre = sin_nada;
        me_llene;
    }
    gritar(cobrar(max / 2 % 5 - 1, 1.5e1), max != 3, max > 1);
}
